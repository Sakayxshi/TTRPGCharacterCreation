    users {
        INT user_id PK
        VARCHAR username UK
        TIMESTAMP created_at
    }
    game_sessions {
        INT session_id PK
        CHAR join_code UK
        ENUM game_type
        VARCHAR session_name
        INT gm_id FK
        TIMESTAMP created_at
    }
    session_players {
        INT user_id PK,FK
        INT session_id PK,FK
        TIMESTAMP joined_at
    }
    characters {
        INT char_id PK
        INT owner_id FK
        INT session_id FK
        VARCHAR name
        SMALLINT age
        VARCHAR gender
        TEXT lore
        BOOLEAN lore_visible
    }
    witcher_characters {
        INT char_id PK,FK

        TEXT lifepath
        TINYINT stat_int_to_stat_luck "9 columns"

        SMALLINT encumbrance "virtual"
    }

        VARCHAR skill_name PK

        TINYINT cost
